# ============================================================
# 已经提交过的，要清史
# ============================================================
#
# 1) 从索引移除（保留本地文件）
#    git rm -r --cached .claude .cursor
#    git commit -m "chore: untrack AI agent config"
#
# 2) 彻底从历史移除（重写历史，慎用，需协调协作者）
#    pip install git-filter-repo
#    git filter-repo --path .claude --path .cursor \
#                    --path AGENTS.md --invert-paths
#
# 3) 若已推到远端且是公开仓库
#    · 视为已泄漏：立刻轮换其中出现过的所有凭据
#    · 强推清理后的历史：git push --force --all
#    · 通知所有协作者重新 clone（历史已变）
#
# 4) 检查历史里还有什么
#    git log --all --full-history -- .claude
#    git rev-list --all | xargs git grep -l 'sk-ant-'   # 扫历史里的 key
#
# ============================================================
# 顺带：检查历史里的密钥
# ============================================================
#
# 装 gitleaks 或 trufflehog 定期扫：
#   gitleaks detect --source . --verbose
#   trufflehog git file://. --only-verified
#
# 也可以加到 pre-commit hook，提交前自动拦
